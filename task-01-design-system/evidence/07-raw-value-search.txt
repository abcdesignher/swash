Command:
$interfaceExtensions = @('*.html','*.css','*.js','*.jsx','*.ts','*.tsx','*.vue','*.svelte')
$files = Get-ChildItem -Path . -Recurse -File -Include $interfaceExtensions -ErrorAction SilentlyContinue | Where-Object { $_.FullName -notmatch '[\\/]\.git[\\/]' -and $_.FullName -notmatch '[\\/]node_modules[\\/]' -and $_.FullName -notmatch '[\\/]task-01-design-system[\\/]styles[\\/]global\.css$' }
Write-Output "Interface source files found:"
$files | ForEach-Object { $_.FullName }
Write-Output "---MATCHES---"
$matches = $files | Select-String -Pattern '#[0-9a-fA-F]{3,8}\b|var\(--primitive-colour' -AllMatches
if ($matches) { $matches } else { Write-Output "(no matches)" }
